## Problem

Clashes between building systems in a 3D model are costly when they are found late. Forge Clash Insight is a prototype for finding them and judging how much each one matters.

## My role

I was the main developer, building it for UtahBIM.

## Approach

- **Detection.** GJK collision detection finds intersecting elements, and a severity score ranks each clash.
- **Sign-in.** Users sign in with Autodesk (APS/ACC) or Procore.
- **Viewer.** Models are processed and shown in a Three.js viewer.
- **Analysis.** LLM analysis of the clashes runs through n8n.

```ts
/**
 * Gilbert-Johnson-Keerthi (GJK) Distance Algorithm
 * Returns true if the two meshes are colliding
 */
export function gjkCollisionDetection(meshA: THREE.Mesh, meshB: THREE.Mesh, maxIterations = 1000): boolean {
  // Convert meshes to convex hulls
  const hullA = meshToConvexHull(meshA);
  const hullB = meshToConvexHull(meshB);

  // Step 1: Pick an arbitrary starting direction
  const direction = new THREE.Vector3(1, 0, 0);

  // Step 2: Get first point in Minkowski difference
  const simplex = new Simplex();
  const initialPoint = supportMinkowski(hullA, hullB, direction);
  simplex.push(initialPoint);

  // Step 4: Choose direction towards origin
  direction.copy(initialPoint).negate();

  // Main loop
  for (let i = 0; i < maxIterations; i++) {
    // Step 5: Get a new point in the direction
    const newPoint = supportMinkowski(hullA, hullB, direction);

    // Step 6: Check if we passed the origin
    if (newPoint.dot(direction) <= 0) {
      // Didn't pass the origin, no collision
      return false;
    }

    // Step 7: Add new point to simplex
    simplex.push(newPoint);

    // Step 8 & 9: Process simplex
    if (processSimplex(simplex, direction)) {
      // Origin is contained in simplex, collision detected
      return true;
    }
  }

  // Max iterations reached (shouldn't happen normally)
  console.warn('GJK: Maximum iterations reached: ' + maxIterations);
  return false;
}
```
